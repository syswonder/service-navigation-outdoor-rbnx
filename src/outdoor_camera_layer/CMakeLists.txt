cmake_minimum_required(VERSION 3.8)
project(outdoor_camera_layer)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(outdoor_camera_layer SHARED src/camera_layer.cpp)
target_compile_features(outdoor_camera_layer PUBLIC cxx_std_17)
ament_target_dependencies(outdoor_camera_layer nav2_costmap_2d pluginlib nav_msgs tf2_ros)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS outdoor_camera_layer LIBRARY DESTINATION lib)
ament_package()
